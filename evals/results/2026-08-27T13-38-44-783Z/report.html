    "s": 1, "sec": 1, "secs": 1, "second": 1, "seconds": 1,
    "m": 60, "min": 60, "mins": 60, "minute": 60, "minutes": 60,

    "d": 86400, "day": 86400, "days": 86400,
}

_PART = re.compile(r"(\d+)\s*([a-z]*)")


def parse_duration(text):
    """Parse a duration such as "90", "30 secs", "1h30m" or "1d 6h 30m" into seconds."""
    if not isinstance(text, str):
        raise ValueError("invalid duration: %r" % (text,))

    s = text.strip().lower()
    total = 0
    pos = 0
    seen = False

    while pos < len(s):
        if s[pos].isspace():
            pos += 1
            continue

        match = _PART.match(s, pos)
        if not match:
            raise ValueError("invalid duration: %r" % (text,))
        value, unit = int(match.group(1)), match.group(2)

        if not unit:
            # A bare number means seconds, but only when it is the whole input.
            if seen or match.end() != len(s):
                raise ValueError("invalid duration: %r" % (text,))
            total += value
        else:
            if unit not in _UNIT_SECONDS:
                raise ValueError("invalid duration: %r" % (text,))
            total += value * _UNIT_SECONDS[unit]

        seen = True
        pos = match.end()

    if not seen:
        raise ValueError("invalid duration: %r" % (text,))
    return total
